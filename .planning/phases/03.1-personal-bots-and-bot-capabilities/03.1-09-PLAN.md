    export const McpTransport = z.enum(["streamable-http", "sse", "sandboxed-stdio"]);
    export const McpHeaderName = z.string().regex(/^[A-Za-z0-9-]{1,64}$/);
    export const McpHeaderInput = z.object({ name: McpHeaderName, value: z.string().min(1).max(4000).optional(), keep: z.boolean().optional() }); // keep = reuse the stored value
    export const McpServerTool = z.object({ name: z.string(), openclawName: z.string(), title: z.string().nullable(), description: z.string().nullable(), readOnly: z.boolean(), destructive: z.boolean() });
    export const McpServer = z.object({ id: Id, name: z.string(), safeName: z.string(), transport: McpTransport, url: z.string().nullable(), sandboxServer: z.string().nullable(),
      headers: z.array(z.object({ name: McpHeaderName, hasValue: z.literal(true) })), orgEnabled: z.boolean(), personalAllowed: z.boolean(), trustedChaining: z.boolean(),
      status: z.enum(["pending", "probing", "ready", "error"]), statusDetail: z.string().nullable(), probedAt: IsoDateTime.nullable(), tools: z.array(McpServerTool),
      createdAt: IsoDateTime, updatedAt: IsoDateTime });
    export const McpServerList = z.object({ servers: z.array(McpServer), sandbox: z.object({ available: z.boolean(), servers: z.array(z.string()) }) });
    export const CreateMcpServerRequest = z.object({ name: z.string().trim().min(1).max(80), transport: McpTransport, url: z.string().max(2000).optional(),
      sandboxServer: z.string().max(64).optional(), headers: z.array(McpHeaderInput).max(8).default([]), orgEnabled: z.boolean().default(true),
      personalAllowed: z.boolean().default(false), trustedChaining: z.boolean().default(false) });
    export const UpdateMcpServerRequest = CreateMcpServerRequest.partial().omit({ transport: true });

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| admin → MCP registry (URLs, headers) | Admin-supplied endpoints and credentials |
| API → cell (`/cell` mcpServers with header values) | Secrets cross to the cell over the authenticated channel |
| MCP server → probe result (tool names, descriptions, annotations) | Third-party metadata is untrusted |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-03.1-09-01 | Information disclosure | MCP credentials | high | mitigate | Stored only in `app.secrets` (AES-GCM, kind `mcp_credential`); DTOs expose `hasValue` only; audits list header names; test greps the value out of response, audit and event |
| T-03.1-09-02 | Server-side request forgery | admin-supplied MCP URLs | medium | mitigate | URL policy (https, or http to compose/RFC 1918 hosts; never loopback, link-local or metadata); the API never connects (probe runs in the cell) |
| T-03.1-09-03 | Elevation of privilege | non-admin registry changes | high | mitigate | `mcp.manage` (admin surface, non-delegable) on every route; 403 test |
| T-03.1-09-04 | Tampering | malicious tool metadata (names, annotations) | medium | mitigate | Names sanitised and length-capped; dangerous names refused by the capability model; annotations only make tools more restricted by default (unknown → side-effecting) |
| T-03.1-09-05 | Elevation of privilege | personal bots using unapproved servers | high | mitigate | `personal_allowed` defaults false and is enforced in the pure decision; test |
| T-03.1-09-06 | Information disclosure | cross-tenant registry | high | mitigate | Tenant RLS on every table; cross-tenant 404 test |
| T-03.1-09-07 | Repudiation | registry changes | medium | mitigate | `mcp.server.created/updated/deleted/probe_requested/probed` audits |
</threat_model>
