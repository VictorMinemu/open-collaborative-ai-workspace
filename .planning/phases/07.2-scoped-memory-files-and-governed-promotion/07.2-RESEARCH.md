## Summary

Today's bot memory is a list of ≤500-char notes per (bot, human) in `app.agent_memories`
[VERIFIED: packages/db/src/schema/app.ts:693-711], written only by the cell tool `ocw_memory_save` through
`POST /agent-channel/v1/memories`, only in a 1-human DM, refused on a tainted turn, injected as ≤50 notes into DM turns,
listed/cleared only by that human, and never readable by admins [VERIFIED: apps/api/src/modules/bot-capabilities/memory.ts:8-16,84-197;
turn-policy.ts:80-84,307-314; routes.ts:337-391; docs/adr/0010…:90-95]. Everything this phase needs already has a
precedent: the pure resolver with relations/preconditions/rules (organization/permissions/*), the unit-persona
visibility rule, STORED tsvector FTS over three unaccented configs, Agent Cards with a single server-decided decision,
the tool gate with taint, and the plugin-tool → Agent Channel route pattern.

Two gaps decide the design. (1) **Approvers usually cannot read the conversation where a proposal is born** (a DM
between a member and their bot), and the card decision route requires `access.readable` of that conversation
[VERIFIED: apps/api/src/modules/chat/card-actions.ts:284-295,360-372]; notifications are message-bound
[VERIFIED: packages/contracts/src/notifications.ts:44-54]. So the primitive must own its decision route and an
approvals inbox, with the chat card as a mirror (and as the decision surface when the approver can read it, e.g. the
owner approving a tainted-turn proposal into their own personal scope). (2) **Agent reads of shared memory leak to the
conversation's audience**: PRD §Shared AI chat context rule says retrieved content "must be authorised for every
participant or is withheld", and "user-private memory never enters a conversation with more than one human"
[CITED: docs/prd/PRD.md:3666-3668]. So agent reads are decided as agent ∩ asking human ∩ every reader of the turn's
conversation.

**Primary recommendation:** Build `memory_scopes` + `memory_files` + append-only `memory_file_revisions` +
`memory_proposals` + a generic `approval_requests` table; add `path_prefix` to `permission_rules` and a `memory`
resource type to the pure resolver; implement approve-and-apply as a new `approval` API module with a kind registry
(`server_apply` mode now, `grant` mode for 07.5), a hash bound to a kind-defined tuple, and its own decision route;
use `@headless-tree/react` (not react-arborist) and plain CodeMirror 6 packages; keep `ocw_memory_save` as a
compatibility alias that appends to the personal `MEMORY.md`.

### Pattern 1: Schema (MEM-01)

Recommended tables (all in `app`, all with `organization_id` → automatic ENABLE+FORCE RLS + `tenant_isolation`
[VERIFIED: packages/db/sql/post-migrate.sql:31-52]):

```ts
// packages/db/src/schema/app.ts (sketch; enum values here are NEW, chosen by this research)
memory_scopes: id, organization_id, kind text check in ('personal','agent','unit','org'),
  agent_id uuid null → agents ON DELETE CASCADE,       // personal + agent
  owner_user_id uuid null,                              // personal only (the human)
  unit_id uuid null → org_units ON DELETE CASCADE,       // unit only
  created_at
  check: personal ⇒ agent_id & owner_user_id not null, unit_id null; agent ⇒ agent_id only;
         unit ⇒ unit_id only; org ⇒ all null
  partial unique: (organization_id, agent_id, owner_user_id) where kind='personal';
                  (organization_id, agent_id) where kind='agent'; (organization_id, unit_id) where kind='unit';
                  (organization_id) where kind='org'
memory_files: id, organization_id, scope_id → memory_scopes CASCADE,
  personal_owner_user_id uuid null   // denormalized for the restrictive RLS policy (Pitfall 3)
  path text, content text, version int default 1, bytes int, deleted_at null,
  updated_by_kind text ('user','agent','system'), updated_by_id uuid, on_behalf_of_user_id uuid null, created_at, updated_at
  unique (scope_id, lower(path)) where deleted_at is null      // case-insensitive uniqueness
  check octet_length(content) <= 262144; check char_length(path) <= 255
  search_tsv: hand-written STORED generated column (ADR-0016 configs) over path + content, GIN index
memory_file_revisions: id, organization_id, file_id → memory_files CASCADE, scope_id, personal_owner_user_id,
  version int, op ('create','update','append','rename','delete','restore'), path, content,
  author_kind, author_id, on_behalf_of_user_id, approval_request_id null, created_at; unique (file_id, version)
memory_proposals: id, organization_id, scope_id, file_id null, path, op ('create','replace','append'),
  base_version int null, content text, content_sha256 text, findings jsonb, reader_count int, tainted bool,
  requested_by_agent_id, requested_for_user_id, event_id, conversation_id, approval_request_id, created_at
approval_requests: id, organization_id, kind text, resource_type text, resource_id uuid,
  action_hash text, status ('pending','approved','denied','applied','failed','expired'),
  mode ('server_apply','grant'), requested_by_agent_id null, requested_for_user_id null, event_id null,
```

#### Path rules (contracts, `parseMemoryPath`) — all limits are [ASSUMED] defaults
- Repo-style relative POSIX path, NFC-normalized, segments split on `/`; reject (never silently fix): leading `/`,
  `\`, empty segments (`a//b`), `.` / `..` segments (also `%2e`), any `%`, control characters, trailing `/`,
  leading/trailing spaces in a segment, segments that are only dots.
- Segment charset `[\p{L}\p{N} ._-]`, 1–64 chars; depth ≤ 8; total ≤ 255 chars; must end in `.md` (lowercase).
- Uniqueness is case-insensitive (`lower(path)`), comparisons are case-sensitive on display.
- Folders are derived from paths (no folder rows); "create folder" in the UI = create `folder/INDEX.md` or a first file.
- `MEMORY.md` at the scope root always exists (created with the scope; cannot be deleted or renamed, only edited).
- Limits: file ≤ 256 KiB UTF-8, ≤ 2,000 live files per scope, path-prefix for rules is a **folder** (`hr/`, always ends
  with `/`, matched on segment boundary).
- Mirror `parseMemoryPath` in the plugin protocol with a parity test (as `isAgentCardRoute` is mirrored,
  [VERIFIED: agent-channel-protocol.ts:457 comment]).

API registry (`apps/api/src/modules/approval/registry.ts`):
```ts
export interface ApprovalKind<P> {
  kind: string;                                  // "memory.write", later "tool.call", "work_zone.start"
  mode: "server_apply" | "grant";
  /** Kind-defined, fixed-order tuple: the ONLY input to the action hash (no generic JSON canonicalization needed). */
  hashInput(tx: Tx, req: ApprovalRow): Promise<readonly (string | number | null)[]>;
  /** Re-decided at decision time with live rules (never cached at request time). */
  canDecide(tx: Tx, deciderUserId: string, req: ApprovalRow): Promise<boolean>;
  /** server_apply only: runs in the decision transaction; returns a result code, never throws for domain conflicts. */
  apply?(tx: Tx, req: ApprovalRow, ctx: { deciderUserId: string; after: AfterCommit }):
    Promise<{ ok: true } | { ok: false; code: string }>;
  /** Detail for the approvals inbox/proposal page, filtered for the viewer. */
  view(tx: Tx, req: ApprovalRow, viewerUserId: string): Promise<P>;
}
export const actionHash = (kind: string, tuple: readonly unknown[]) =>
  sha256Hex(JSON.stringify(["ocw.approval.v1", kind, ...tuple]));  // tuple order fixed by the kind
```

3. `kind.canDecide` (resolver, live rules) → 403 `not_approver`.
4. Recompute `actionHash(kind, await kind.hashInput(...))`; must equal the stored hash **and** the request's

5. `UPDATE … SET status=$outcome, decided_by, decided_at WHERE id=$1 AND status='pending'` (0 rows → 409

6. `server_apply` + approved: consume the one-use grant in the same statement family —

7. Audit `approval.decided` (+ `approval.applied|approval.apply_failed`) with ids, kind, outcome, result code; mirror the

### Anti-Patterns to Avoid
- **Using the cell's file memory or `read/write/edit/ls`** — stays off (D-01; render-config denies them).
- **Model-supplied owner, scope readers or card fields** — the API composes the approval card (platform-owned).
- **Pushing memory content or paths on `org:<id>`** — never; ids-only events on `user:<id>` streams.
- **Generic JSON canonicalization in 07.2** — hash a kind-defined tuple; leave RFC 8785 to 07.5.
- **Checking only scope-level permission for a file** — every file decision carries `path`.
- **Adding `channel` memory** — PRD §81 lists Channel Memory, but D-01 does not; out of scope.

## Common Pitfalls

### Pitfall 1: Concurrent edits (human ↔ human, human ↔ agent)
**What goes wrong:** last write wins silently. **How to avoid:** `UPDATE … WHERE id=$1 AND version=$expected`,
409 with current version; agents use `append` or `expectedVersion`; proposals carry `base_version` and fail as
`stale_base` at apply time (card shows "file changed since the proposal"; never auto-rebase). **Warning sign:** a
revision whose parent version ≠ previous revision version.

### Pitfall 2: Large files
**What goes wrong:** 256 KiB documents in turn context, slow Markdown render, huge jsonb. **How to avoid:** DB check
constraint + Zod limit; tree/list endpoints return metadata only; turn injection truncates `MEMORY.md` to the budget
with a marker telling the model to use `ocw_memory_read`; proposals store content in `memory_proposals`, not in
`approval_requests`.

### Pitfall 3: Admin readability of personal memory (ADR-0010)
**What goes wrong:** an admin path, a search query, an audit row or a reader-count call reveals personal content or
paths. **How to avoid:** resolver precondition (owner-only, admins included), restrictive RLS on personal rows, audit
metadata with ids/versions/bytes only (no paths — paths are content), governance views never list personal scopes.
**Test:** admin session gets 404 on personal file ids, 0 rows via raw SQL, no personal hits in search.

### Pitfall 4: Path traversal / ambiguity in a virtual FS
**What goes wrong:** `a/../b.md`, `%2e%2e`, NFD vs NFC duplicates, `Hr/` vs `hr/` dodging a deny rule, prefix `hr`
matching `hr-public/`. **How to avoid:** strict `parseMemoryPath` (reject, don't fix), NFC, case-insensitive uniqueness,
rules on folder prefixes with trailing `/` compared case-insensitively on segment boundaries; the Next.js catch-all
`[...path]` segments are re-joined and re-parsed. **Test:** property-style table of rejects; deny `hr/` beats any casing.

### Pitfall 5: Leak through an agent reply (audience)
**What goes wrong:** a member's bot reads team memory in a public channel and quotes it. **How to avoid:** agent reads =
agent ∩ asking human ∩ every human reader of the turn's conversation: 1-human DM → the human; private channel/group/
listed AI-chat readers → each must hold `memory.read` on the file (batch decide); public channel or a group-granted AI
chat (unbounded audience) → only org-scope files nobody is denied; personal scope only in 1-human DM (today's rule,
[VERIFIED: turn-policy.ts:307-314]); refuse with `audience_too_wide` (pattern `TurnWithheldReason`
[VERIFIED: bot-capabilities.ts:245-250]). Source: PRD rule [CITED: docs/prd/PRD.md:3666-3668].

### Pitfall 6: Cross-unit leak
**What goes wrong:** sibling team or parent-only member reads a team's memory; a unit lead of team A promotes into team
B. **How to avoid:** unit-persona rule (member of subtree / lead on path / admin), `memory.promote` from
`LEAD_PERMISSIONS` only on the lead's subtree. **Test matrix:** dept D ⊃ teams A, B: A-member reads A and D, not B;
D-only member reads D, not A; D-lead reads/promotes A and B; A-lead cannot promote B.

### Pitfall 7: Leak through search or realtime
**What goes wrong:** FTS counts/snippets from unreadable files; `org:` events with paths. **How to avoid:** Pattern 7;
realtime only `memory.changed {scopeId, fileId, version}` on the personal owner's `user:` stream and
`approval_request.updated {id, status}` on deciders'/requester's `user:` streams; shared-scope viewers refetch on focus
and rely on 409.

### Pitfall 8: Secrets written by humans
**What goes wrong:** detector only on proposals; a pasted API key lands in DB plaintext (principle 8). **How to avoid:**
secret layer on every write path (UI, `write`, `propose`, legacy `save`, migration excluded); 422 with masked samples.

### Pitfall 9: Stateful global regexes
**What goes wrong:** shared `/g` RegExp objects keep `lastIndex` across calls (sanitize.ts resets it manually
[VERIFIED: sanitize.ts:88]) → intermittent misses when detectors reuse them. **How to avoid:** `matchAll` on fresh
`new RegExp(src, flags)` or reset `lastIndex` in every function; test the same input twice.

### Pitfall 10: Agent rewrites the approval card
**What goes wrong:** `PATCH /agent-channel/v1/cards/:id` lets the posting agent change fields/summary
[VERIFIED: agent-channel/cards.ts:234-296]. **How to avoid:** `platform_owned` flag, 403 for agent patches, test.

### Pitfall 11: Parallel phases colliding
**What goes wrong:** 07.1/07.3/07.4 also add migrations (drizzle snapshot chain, `HEAD_ONLY`), ADR numbers, and
permission enum values. **How to avoid:** generate the migration on top of the latest `main` at execution time; take the
next free ADR number; append-only enum edits.

### Checksums
```ts
export function luhnOk(digits: string): boolean {
  let sum = 0; let dbl = false;
  for (let i = digits.length - 1; i >= 0; i--) {
    let d = digits.charCodeAt(i) - 48; if (dbl) { d *= 2; if (d > 9) d -= 9; } sum += d; dbl = !dbl;
  }
  return digits.length >= 13 && sum % 10 === 0;
}
export function ibanOk(raw: string): boolean {
  const s = raw.replace(/\s+/g, "").toUpperCase();
  if (!/^[A-Z]{2}\d{2}[A-Z0-9]{11,30}$/.test(s)) return false;
  const moved = s.slice(4) + s.slice(0, 4);
  let rem = 0;
  for (const ch of moved) { const v = ch >= "A" ? String(ch.charCodeAt(0) - 55) : ch;
    for (const c of v) rem = (rem * 10 + (c.charCodeAt(0) - 48)) % 97; }
  return rem === 1;
}
const NIF_LETTERS = "TRWAGMYFPDXBNJZSQVHLCKE";
export const nifOk = (s: string) => /^\d{8}[A-Z]$/.test(s) && NIF_LETTERS[Number(s.slice(0, 8)) % 23] === s[8];
export const nieOk = (s: string) =>
  /^[XYZ]\d{7}[A-Z]$/.test(s) && nifOk(`${"XYZ".indexOf(s[0] as string)}${s.slice(1)}`.padStart(9, "0"));
```

### Presidio call with timeout, fail-open marker

```ts
export async function presidioFindings(env: Env, text: string, log: Logger): Promise<Finding[]> {
  if (!env.OCW_PRESIDIO_URL) return [];
  try {
    const out = await Promise.all(env.OCW_PRESIDIO_LANGUAGES.map(async (language) => {
      const res = await fetch(new URL("/analyze", env.OCW_PRESIDIO_URL), {
        method: "POST", headers: { "content-type": "application/json" },
        body: JSON.stringify({ text, language, entities: PRESIDIO_ENTITIES, score_threshold: 0.5 }),
        signal: AbortSignal.timeout(env.OCW_PRESIDIO_TIMEOUT_MS),
      });
      if (!res.ok) throw new Error(`presidio ${res.status}`);
      return PresidioResult.parse(await res.json());         // Zod at the boundary
    }));
    return toFindings(out.flat(), text);
  } catch (err) {
    log.warn({ err: (err as Error).name }, "presidio unavailable");   // never the text
    return [{ kind: "extended_scan_unavailable", severity: "info", count: 0, samples: [], lines: [] }];
  }
}
```
