    -- post-migrate.sql (idempotent: DROP POLICY IF EXISTS … then CREATE)
    audience_backstop ON app.work_sessions AS RESTRICTIVE FOR ALL
      USING  (user-less OR cardinality(app.work_session_access(id, current user)) > 0)
      WITH CHECK (user-less OR created_by = current user OR cardinality(app.work_session_access(id, current user)) > 0)
    audience_backstop ON app.work_prompts / app.work_approvals / app.work_session_grants AS RESTRICTIVE FOR ALL
      USING / WITH CHECK (EXISTS (SELECT 1 FROM app.work_sessions s WHERE s.id = <table>.work_session_id))
    -- "user-less" = nullif(current_setting('app.user_id', true), '') IS NULL
