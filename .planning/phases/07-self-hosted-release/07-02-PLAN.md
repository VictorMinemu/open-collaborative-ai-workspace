    // packages/db/src/schema-check.ts (exported from packages/db/src/index.ts)
    export type SchemaStatus = "ok" | "behind" | "ahead" | "diverged" | "missing";
    export interface SchemaCheck { status: SchemaStatus; pending: string[]; unknown: string[]; modified: string[] }
    export function compareMigrations(local: {name: string; hash: string}[], applied: {name: string; hash: string}[]): SchemaCheck;
    export async function checkSchema(databaseUrl: string, migrationsDir?: string): Promise<SchemaCheck>; // permission denied / no table → "missing"
    export function describeSchemaMismatch(c: SchemaCheck): string;
    //   behind  → "Database schema is behind this API (N pending, newest X). Run: docker compose run --rm migrate"
    //   ahead   → "Database was migrated by a newer release (unknown: Y). Restore a backup or run the newer image."
    //   diverged→ "Migration X was modified after it was applied."
    //   missing → "Database has never been migrated (or ocw_app cannot read drizzle.__drizzle_migrations). Run: docker compose run --rm migrate"

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|

| T-07-02-02 | Elevation of privilege | new grant to `ocw_app` | low | mitigate | USAGE on `drizzle` + SELECT on one table only; int test proves no write (Task 1) |
